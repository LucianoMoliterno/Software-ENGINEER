PUNTO 5 - publicar eventos (PROBAR CON FECHA FUTURA):
# Publicar evento con fecha de DICIEMBRE 2025 (futuro real)
echo '{"idOrganizacion":456,"idEvento":1001,"nombreEvento":"Campaña de Invierno 2025","descripcion":"Recolección de ropa","fechaHora":"2025-12-15T10:00:00"}' | docker exec -i kafka kafka-console-producer --bootstrap-server localhost:9092 --topic eventos-solidarios

# Esperar 3 segundos
Start-Sleep -Seconds 3

# Verificar que se guardó
Invoke-RestMethod http://localhost:8082/externos
